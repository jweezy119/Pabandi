
// PABANDI AI AGENT MARKETPLACE — self-healing profitable loop

model AgentProfile {
  id                String   @id @default(cuid())
  name              String
  slug              String   @unique
  description       String
  capabilities      String[]
  walletAddress     String
  publicKey         String
  reputation        Float    @default(0)
  totalEarned       Float    @default(0)
  totalSpent        Float    @default(0)
  projectsCompleted Int      @default(0)
  projectsFailed    Int      @default(0)
  isActive          Boolean  @default(true)
  metadata          Json?
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt

  postedProjects       AgentProject[]           @relation("postedProjects")
  bids                 AgentProjectBid[]        @relation("AgentProjectBid_bidder")
  sentTransactions     AgentMarketTransaction[] @relation("sentTransactions")
  receivedTransactions AgentMarketTransaction[] @relation("receivedTransactions")

  @@index([walletAddress])
  @@index([slug])
}

model AgentProject {
  id            String   @id @default(cuid())
  title         String
  description   String
  requirements  String
  posterId      String
  poster        AgentProfile @relation("postedProjects", fields: [posterId], references: [id])
  budgetUsd     Float
  budgetPab     Float
  deadline      DateTime
  status        String   @default('OPEN')
  selectedBidId String?
  escrowId      String?
  category      String
  complexity    String   @default('MEDIUM')
  metadata      Json?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  bids   AgentProjectBid[]
  escrow AgentEscrow?

  @@index([status])
  @@index([category])
  @@index([posterId])
}

model AgentProjectBid {
  id             String   @id @default(cuid())
  projectId      String
  project        AgentProject @relation(fields: [projectId], references: [id], onDelete: Cascade)
  bidderId       String
  bidder         AgentProfile @relation("AgentProjectBid_bidder", fields: [bidderId], references: [id])
  proposedAmount Float
  proposedPab    Float
  timelineHours  Int
  approach       String
  status         String   @default('PENDING')
  isWinning      Boolean  @default(false)
  submittedAt    DateTime @default(now())
  acceptedAt     DateTime?

  @@index([projectId])
  @@index([bidderId])
}

model AgentEscrow {
  id            String   @id @default(cuid())
  projectId     String   @unique
  project       AgentProject @relation(fields: [projectId], references: [id])
  totalAmount   Float
  releaseAmount Float
  platformFee   Float
  status        String   @default('FUNDED')
  fundedAt      DateTime @default(now())
  releasedAt    DateTime?
  refundedAt    DateTime?
}

model AgentMarketTransaction {
  id             String   @id @default(cuid())
  projectId      String
  fromAgentId    String
  fromAgent      AgentProfile @relation("sentTransactions", fields: [fromAgentId], references: [id])
  toAgentId      String
  toAgent        AgentProfile @relation("receivedTransactions", fields: [toAgentId], references: [id])
  amount         Float
  pabReward      Float
  platformFeeUsd Float
  platformFeeSol Float
  type           String
  status         String   @default('COMPLETED')
  txHash         String?
  createdAt      DateTime @default(now())

  @@index([fromAgentId])
  @@index([toAgentId])
}
